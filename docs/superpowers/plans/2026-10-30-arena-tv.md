**Goal:** Turn the landing page into an instant live broadcast of the real exhibition fight, with a cinematic director, real scoreboard/kill feed, template commentary, and an honest highlights fallback.

Spectators reuse the existing `MGS_SPECTATOR_CAP`; over-cap/failure → existing highlight-video hero.

**Tech Stack:** Rust (axum headers, existing `main_tests.rs`), vanilla JS ES modules (`static_client/client_logic/`, `node --test`), landing `website/js/main.js`, Playwright for e2e.

```js
// static_client/client_logic/commentary_templates.js
// Event -> broadcast commentary lines. Every field printed comes from the
// event payload; an event missing its fields yields no line (never
// "undefined"/"NaN" on air). Cooldowns per type, dedup per exact line,
// deterministic pool rotation.

export const COMMENTARY_POOLS = Object.freeze({
    streak: Object.freeze([
        (e) => `${e.name} is on a ${e.count}-kill streak`,
        (e) => `${e.count} straight for ${e.name} — who stops them?`,
        (e) => `${e.name} heating up: ${e.count} kills without dying`,
    ]),
    multi_kill: Object.freeze([
        (e) => `Multi-kill x${e.count} by ${e.name}!`,
        (e) => `${e.name} deletes ${e.count} fighters in one push`,
        (e) => `${e.count} down at once — ${e.name} on a rampage`,
    ]),
    low_hp_escape: Object.freeze([
        (e) => `${e.name} survives on a sliver of HP`,
        (e) => `One hit from death, ${e.name} gets out`,
        (e) => `${e.name} refuses to die`,
    ]),
    lead_change: Object.freeze([
        (e) => `Team ${e.team} takes the lead`,
        (e) => `Momentum swing — team ${e.team} in front now`,
        (e) => `Team ${e.team} flips the scoreboard`,
    ]),
    champion_kill: Object.freeze([
        (e) => `${e.killer} takes down the champion ${e.victim}`,
        (e) => `Champion falls: ${e.killer} gets ${e.victim}`,
        (e) => `${e.victim} dethroned by ${e.killer}`,
    ]),
    duel_start: Object.freeze([
        (e) => `${e.name} steps up to challenge the champion`,
        (e) => `A wild challenger appears: ${e.name}`,
        (e) => `${e.name} wants the crown`,
    ]),
});

const REQUIRED_FIELDS = {
    streak: ['name', 'count'],
    multi_kill: ['name', 'count'],
    low_hp_escape: ['name'],
    lead_change: ['team'],
    champion_kill: ['killer', 'victim'],
    duel_start: ['name'],
};

export function createCommentary({ cooldownMs = 20000, dedupMs = 60000 } = {}) {
    const lastTypeAt = new Map();
    const recentLines = new Map();
    const counters = new Map();

    function onEvent(ev, atMs) {
        if (!ev || !COMMENTARY_POOLS[ev.type]) return null;
        const fields = REQUIRED_FIELDS[ev.type] || [];
        for (const f of fields) {
            const v = ev[f];
            if (v === undefined || v === null || v === '' || (typeof v === 'number' && !Number.isFinite(v))) return null;
        }
        const last = lastTypeAt.get(ev.type);
        if (last !== undefined && atMs - last < cooldownMs) return null;
        const i = (counters.get(ev.type) || 0) % COMMENTARY_POOLS[ev.type].length;
        const line = COMMENTARY_POOLS[ev.type][i](ev);
        const seenAt = recentLines.get(line);
        if (seenAt !== undefined && atMs - seenAt < dedupMs) return null;
        counters.set(ev.type, (counters.get(ev.type) || 0) + 1);
        lastTypeAt.set(ev.type, atMs);
        recentLines.set(line, atMs);
        for (const [l, t] of recentLines) if (atMs - t > dedupMs) recentLines.delete(l);
        return line;
    }

    return { onEvent };
}
```

```js
// static_client/client_logic/BroadcastDirector.js
// Cinematic camera target selection for Arena TV. Pure logic, no PIXI/DOM:
// the caller applies { focusX, focusY, zoom } to the scene each frame.
// Scoring uses real damage events (ids) and kill-feed entries (names —
// the wire protocol carries no kill ids).

export function createBroadcastDirector({
    dwellMs = 4000,
    switchMargin = 1.35,
    switchSustainMs = 1500,
    damageDecayMs = 5000,
    killBoostMs = 10000,
    killZoomMs = 1200,
} = {}) {
    const damage = new Map(); // playerId -> [{ value, atMs }]
    const kills = new Map(); // username -> [atMs]
    let targetId = null;
    let targetSince = 0;
    let challengerId = null;
    let challengerSince = 0;
    let zoomUntil = 0;

    function recordDamage({ instigatorId, value, atMs }) {
        if (!instigatorId || !Number.isFinite(value)) return;
        const list = damage.get(instigatorId) || [];
        list.push({ value, atMs });
        damage.set(instigatorId, list);
    }

    function recordKill({ killerName, atMs }) {
        if (!killerName) return;
        const list = kills.get(killerName) || [];
        list.push(atMs);
        kills.set(killerName, list);
        const target = currentTargetName;
        if (target && killerName === target) zoomUntil = atMs + killZoomMs;
    }

    let currentTargetName = null;

    function recentDamage(id, atMs) {
        const list = (damage.get(id) || []).filter((d) => atMs - d.atMs <= damageDecayMs);
        damage.set(id, list);
        return list.reduce((s, d) => s + d.value, 0);
    }

    function recentKills(name, atMs) {
        const list = (kills.get(name) || []).filter((t) => atMs - t <= killBoostMs);
        kills.set(name, list);
        return list.length;
    }

    function scoreOf(p, players, leaderId, atMs) {
        let score = 3 * recentDamage(p.id, atMs) + 8 * recentKills(p.username, atMs);
        if (p.id === leaderId) score += 0.5;
        let nearestEnemy = Infinity;
        for (const q of players) {
            if (q.id === p.id || q.teamId === p.teamId) continue;
            const dist = Math.hypot(q.x - p.x, q.y - p.y);
            if (dist < nearestEnemy) nearestEnemy = dist;
        }
        if (nearestEnemy < 400) score += 1.5;
        if (p.maxHp > 0 && p.hp / p.maxHp < 0.3 && nearestEnemy < 600) score += 2;
        return score;
    }

    function selectTarget(players, leaderId, atMs) {
        if (!players.length) {
            targetId = null;
            currentTargetName = null;
            return null;
        }
        const scored = players.map((p) => ({ p, s: scoreOf(p, players, leaderId, atMs) }));
        scored.sort((a, b) => b.s - a.s || (a.p.id < b.p.id ? -1 : 1)); // deterministic tie-break
        const best = scored[0];
        if (targetId === null || !players.some((p) => p.id === targetId)) {
            targetId = best.p.id;
            targetSince = atMs;
            challengerId = null;
        } else if (best.p.id !== targetId) {
            const current = scored.find((e) => e.p.id === targetId);
            const currentScore = current ? current.s : 0;
            const dwellOver = atMs - targetSince >= dwellMs;
            if (dwellOver && best.s > Math.max(currentScore, 0.01) * switchMargin) {
                if (challengerId !== best.p.id) {
                    challengerId = best.p.id;
                    challengerSince = atMs;
                } else if (atMs - challengerSince >= switchSustainMs) {
                    targetId = best.p.id;
                    targetSince = atMs;
                    challengerId = null;
                }
            } else {
                challengerId = null;
            }
        } else {
            challengerId = null;
        }
        const target = players.find((p) => p.id === targetId);
        currentTargetName = target ? target.username : null;
        return targetId;
    }

    function cameraFrame(players, leaderId, atMs) {
        if (!players.length) return null;
        const id = selectTarget(players, leaderId, atMs);
        const target = players.find((p) => p.id === id);
        if (!target) return null;
        return { focusX: target.x, focusY: target.y, zoom: atMs < zoomUntil ? 1.15 : 1, targetId: id };
    }

    return { recordDamage, recordKill, selectTarget, cameraFrame };
}
```

Note: the `switch after dwell` test drives damage at 5100 with decay 5000 — by 6700 the event is still inside the decay window, and `a`'s score is 0, so the margin check uses the 0.01 floor. The `dwell` test relies on the initial lock happening at 1000 with no prior state. Do not reorder these operations when implementing.

```js
// static_client/client_logic/BroadcastOverlay.js
// DOM overlay for Arena TV: LIVE badge, team scores, match clock, kill
// feed, commentary line. Kill-feed/clock logic is pure and unit-tested;
// DOM assembly is a thin shell over it. All values come from the server
// (scoreboard endpoint, kill-feed events) — nothing is invented here.

export const KILL_FEED_TTL_MS = 8000;
export const KILL_FEED_LIMIT = 5;

export function pushKillEntry(feed, entry) {
    return [entry, ...feed].slice(0, KILL_FEED_LIMIT);
}

export function pruneKillFeed(feed, atMs, ttlMs = KILL_FEED_TTL_MS) {
    return feed.filter((e) => atMs - e.atMs <= ttlMs);
}

export function formatClock(seconds) {
    if (!Number.isFinite(seconds)) return '--:--';
    const s = Math.max(0, Math.floor(seconds));
    return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, '0')}`;
}

const esc = (s) => String(s).replace(/[&<>"']/g, (c) => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;',
}[c]));

export function createBroadcastOverlay({ document: doc, parent }) {
    const element = doc.createElement('div');
    element.className = 'arena-tv-overlay';
    element.innerHTML =
        '<div class="arena-tv-top">' +
        '<span class="arena-tv-live">LIVE</span>' +
        '<span class="arena-tv-scores"></span>' +
        '<span class="arena-tv-clock">--:--</span>' +
        '</div>' +
        '<div class="arena-tv-killfeed"></div>' +
        '<div class="arena-tv-commentary"></div>';
    parent.appendChild(element);
    const scoresEl = element.querySelector('.arena-tv-scores');
    const clockEl = element.querySelector('.arena-tv-clock');
    const feedEl = element.querySelector('.arena-tv-killfeed');
    const commentaryEl = element.querySelector('.arena-tv-commentary');
    let feed = [];

    function renderFeed() {
        feedEl.innerHTML = feed.map((e) =>
            `<div class="arena-tv-kill">${esc(e.killerName)} ⟶ ${esc(e.victimName)}</div>`,
        ).join('');
    }

    return {
        element,
        pushKill(entry) {
            feed = pushKillEntry(feed, entry);
            renderFeed();
        },
        tick(atMs) {
            const before = feed.length;
            feed = pruneKillFeed(feed, atMs);
            if (feed.length !== before) renderFeed();
        },
        setCommentary(line) {
            commentaryEl.textContent = line || '';
        },
        setScoreboard({ teamScores, timeRemaining }) {
            if (Array.isArray(teamScores)) {
                scoresEl.textContent = teamScores.map((s, i) => `T${i + 1} ${s}`).join(' · ');
            }
            clockEl.textContent = formatClock(timeRemaining);
        },
    };
}
```

```js
    const field = document.querySelector('.arena-field');

    function onMessage(ev) {
```
